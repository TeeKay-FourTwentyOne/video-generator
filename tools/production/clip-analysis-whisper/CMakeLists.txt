cmake_minimum_required(VERSION 3.20)
project(clip_analysis_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static audited runtime")
set(GGML_METAL OFF CACHE BOOL "Bounded CPU speech pass")
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "")
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "")
if(NOT IS_DIRECTORY "${WHISPER_SOURCE}")
  message(FATAL_ERROR "Explicitly provision a pinned whisper.cpp checkout first")
endif()
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(clip-analysis-whisper ../clip-analysis-whisper.cpp)
target_include_directories(clip-analysis-whisper PRIVATE "${WHISPER_SOURCE}/examples")
target_link_libraries(clip-analysis-whisper PRIVATE whisper)
